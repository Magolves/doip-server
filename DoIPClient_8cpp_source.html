247 void DoIPClient::tcpThreadFunction() {
248  int sendRetries = 5;
249  int receiveRetries = 5;
250 
251  while (m_tcpRunning.load()) {
252  if (m_messageQueue.empty()) {
253  std::this_thread::sleep_for(std::chrono::milliseconds(100));
254  continue;
255  }
256 
257  DoIPMessage msg;
258  m_messageQueue.pop(msg);
259  if (sendDoIPMessage(msg) < 0) {
260  --sendRetries;
261  if (sendRetries == 0) {
262  m_log->error("Exceeded maximum send retries, close connection");
263  m_tcpRunning.store(false);
264  break;
265  }
266  m_log->error("Failed to send DoIP message from queue, retries left: {}", sendRetries);
267  std::this_thread::sleep_for(std::chrono::milliseconds(100));
268  continue;
269  } else {
270  m_model->messageSent(*this, msg);
271  }
272 
273  sendRetries = 5;
274 
275  // Receive response (ack or alive check)
276  auto optAck = receiveMessage();
277  if (optAck == std::nullopt) {
278  m_log->error("Failed to receive DoIP ack");
279  std::this_thread::sleep_for(std::chrono::milliseconds(100));
280  continue;
281  } else {
282  handleMessage(optAck.value());
283  }
284 
285  if (m_receiveState == ReceiveState::Quit) {
286  m_log->info("Receive state set to Quit, closing TCP thread");
287  break;
288  }
289 
290  if (m_receiveState == ReceiveState::WaitForAckOrAliveCheck) {
291  continue;
292  }
293 
294  auto optMsg = receiveMessage();
295  if (optMsg == std::nullopt) {
296  --receiveRetries;
297  if (receiveRetries == 0) {
298  m_log->error("Exceeded maximum receive retries, close connection");
299  m_tcpRunning.store(false);
300  break;
301  }
302  m_log->error("Failed to receive DoIP message, retries left: {}", receiveRetries);
303  std::this_thread::sleep_for(std::chrono::milliseconds(100));
304  continue;
305  } else {
306  receiveRetries = 5;
307  handleMessage(optMsg.value());
308  }
309  }

312 void DoIPClient::handleMessage(const DoIPMessage &msg) {
313  if (m_receiveState == ReceiveState::WaitForAckOrAliveCheck) {
314  switch (msg.getPayloadType()) {
315  case DoIPPayloadType::AliveCheckRequest:
316  m_log->info("Received Alive Check Request, sending Alive Check Response");
317  if (sendAliveCheckResponse() <= 0) {
318  m_log->error("Failed to send Alive Check Response");
319  m_model->error(*this, "Failed to send Alive Check Response");
320  }
321  break;
322  case DoIPPayloadType::DiagnosticMessageNegativeAck:
323  m_model->diagMessageAcked(*this, msg.getDiagnosticAck().value_or(DoIPDiagnosticAck::TransportProtocolError));
324  break;
325  case DoIPPayloadType::DiagnosticMessageAck:
326  // received diag msg ack -> proceed to receive diag messages
327  updateReceiveState(ReceiveState::WaitForDiagnosticMessage);
328  m_model->diagMessageAcked(*this, DoIPDiagnosticAck::PositiveAck);
329  break;
330  default:
331  m_log->warn("Received unexpected message type ({}) while waiting for Ack or Alive Check", fmt::streamed(msg.getPayloadType()));
332  m_model->error(*this, "Received unexpected message type");
333  break;
334  }
335  } else if (m_receiveState == ReceiveState::WaitForDiagnosticMessage) {
336  // Process diagnostic message
337  auto newState = ReceiveState::WaitForAckOrAliveCheck;
338  auto payloadType = msg.getPayloadType();
339  switch (payloadType) {
340  case DoIPPayloadType::DiagnosticMessage: {
341  auto result = m_model->diagMessageReceived(*this, msg);
342  if (result == CallbackResult::Stop) {
343  newState = ReceiveState::Quit;
344  m_tcpRunning.store(false);
345  }
346  break;
347  }
348  default:
349  m_log->warn("Received unhandled DoIP message type: {}", fmt::streamed(payloadType));
350  m_model->error(*this, "Received unexpected message type");
351  break;
352  }
353  updateReceiveState(newState);
354  }
355 }
