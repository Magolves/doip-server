117 bool DoIPServer::setupUdpSocket() {
118  m_udpLog->debug("Setting up UDP socket for broadcasts");
119 
120  int tmpUdpSocket = socket(AF_INET, SOCK_DGRAM, 0);
121  if (tmpUdpSocket < 0) {
122  m_udpLog->error("Failed to create UDP socket: {}", strerror(errno));
123  return false;
124  }
125 
126  auto port = DOIP_UDP_DISCOVERY_PORT;
127 
128  // Set socket timeout
129  struct timeval timeout;
130  timeout.tv_sec = 1;
131  timeout.tv_usec = 0;
132  setsockopt(tmpUdpSocket, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout));
133 
134  // Enable SO_REUSEADDR
135  int reuse = 1;
136  setsockopt(tmpUdpSocket, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse));
137 
138  // Enable SO_BROADCAST to receive broadcast packets
139  int broadcast = 1;
140  if (setsockopt(tmpUdpSocket, SOL_SOCKET, SO_BROADCAST, &broadcast, sizeof(broadcast)) < 0) {
141  m_udpLog->warn("Failed to enable SO_BROADCAST for receive socket: {}", strerror(errno));
142  }
143 
144  // Bind to discovery port
145  struct sockaddr_in udp_addr;
146  memset(&udp_addr, 0, sizeof(udp_addr));
147  udp_addr.sin_family = AF_INET;
148  udp_addr.sin_addr.s_addr = htonl(INADDR_ANY);
149  udp_addr.sin_port = htons(port);
150 
151  if (bind(tmpUdpSocket, reinterpret_cast<struct sockaddr *>(&udp_addr), sizeof(udp_addr)) < 0) {
152  m_udpLog->error("Failed to bind UDP socket to port {}: {}", port, strerror(errno));
153  ::close(tmpUdpSocket);
154  tmpUdpSocket = -1;
155  return false;
156  }
157 
158  m_udpSocket.reset(tmpUdpSocket);
159 
160  m_udpLog->info("UDP socket bound to port {}", port);
161  m_udpRunning.store(true);
162  m_workerThreads.emplace_back([this]() {
163  udpAnnouncementThread();
164  });
165 
166  return true;
167 }

231  m_doipLog->info("Announcement thread started");
232 
233  if (!m_transport) {
234  m_doipLog->error("No transport available for announcements");
235  return;
236  }
237

240  DoIPMessage msg = makeVehicleAnnouncementMessage(m_config);
241  ssize_t sentBytes = sendBroadcast(msg, DOIP_UDP_TEST_EQUIPMENT_REQUEST_PORT);
242 
243  m_doipLog->info("TX {}", fmt::streamed(msg));
244  if (sentBytes > 0) {
245  m_udpLog->info("Sent Vehicle Announcement: {} bytes", sentBytes);
246  } else {
247  m_udpLog->error("Failed to send announcement");
248  }
249

251 
252  usleep(m_config.announceInterval * 1000);
253  }

256  while(m_udpRunning.load()) {

258  struct sockaddr_in clientAddr;
259  socklen_t clientAddrLen = sizeof(clientAddr);
260 
261  ssize_t result = recvfrom(m_udpSocket.get(), receive.data(), DOIP_MAXIMUM_MTU, 0,
262  reinterpret_cast<struct sockaddr*>(&clientAddr), &clientAddrLen);
263 
264  if (result <= 0) {
265  if (result < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
266  m_udpLog->warn("Receive error: {} ({})", strerror(errno), errno);
267  }
268  std::this_thread::sleep_for(std::chrono::milliseconds(100));
269  continue;
270  }
271

273  if (!optMsg.has_value()) {

275  m_udpLog->error("Received invalid message");
276  continue;
277  }
278 
279  const auto& msg = optMsg.value();

281  inet_ntoa(clientAddr.sin_addr), ntohs(clientAddr.sin_port));
282

293  }
